/* 구글 앱스크립트 (Code.gs) 파일 내용 */
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var wordSheet = ss.getSheetByName("WordList");
  var recordSheet = ss.getSheetByName("StudentRecords");
  if (!wordSheet) setupSheets();

  var action = e ? e.parameter.action : null;
  if (action === "getWords") {
    var data = wordSheet.getDataRange().getValues();
    data.shift();
    return responseJSON({ success: true, words: data });
  }
  if (action === "getRankings") {
    var records = recordSheet.getDataRange().getValues();
    records.shift();
    records.sort(function(a, b) { return Number(b[2]) - Number(a[2]); });
    return responseJSON({ success: true, rankings: records.slice(0, 50) });
  }
  return responseJSON({ success: true, message: "Apps Script running!" });
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var recordSheet = ss.getSheetByName("StudentRecords");
    if (!recordSheet) setupSheets();

    var data = JSON.parse(e.postData.contents);
    recordSheet.appendRow([
      new Date().toLocaleString("ko-KR"),
      data.studentName,
      data.score,
      data.accuracy + "%",
      data.difficulty,
      data.correctCount,
      data.wrongCount
    ]);
    return responseJSON({ success: true });
  } catch (err) {
    return responseJSON({ success: false, error: err.toString() });
  }
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function setupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var wordSheet = ss.getSheetByName("WordList") || ss.insertSheet("WordList");
  if (wordSheet.getLastRow() === 0) {
    wordSheet.appendRow(["현재형(Present)", "과거형(Correct)", "오답1", "오답2", "오답3"]);

    wordSheet.appendRow(["run", "ran", "runned", "runed", "ron"]);
    wordSheet.appendRow(["eat", "ate", "eated", "eaten", "aet"]);
    wordSheet.appendRow(["play", "played", "playied", "playd", "pleyed"]);
    wordSheet.appendRow(["make", "made", "maked", "maded", "mook"]);
    wordSheet.appendRow(["buy", "bought", "buyed", "bougth", "boated"]);
    wordSheet.appendRow(["study", "studied", "studyed", "studid", "studiied"]);
    wordSheet.appendRow(["see", "saw", "seed", "seen", "so"]);
    wordSheet.appendRow(["take", "took", "taked", "taken", "tuke"]);
    wordSheet.appendRow(["write", "wrote", "writed", "written", "wroted"]);
  }
  var recordSheet = ss.getSheetByName("StudentRecords") || ss.insertSheet("StudentRecords");
  if (recordSheet.getLastRow() === 0) {
    recordSheet.appendRow(["일시", "학생 이름/학번", "최종 점수", "정답률", "난이도(목숨)", "맞힌 개수", "틀린 개수"]);
  }

        class SoundEffects {
            constructor() {
                this.ctx = null;
                this.muted = false;
            }

            init() {
                if (!this.ctx) {
                    const AudioCtx = window.AudioContext || window.webkitAudioContext;
                    if (AudioCtx) this.ctx = new AudioCtx();
                }
                if (this.ctx && this.ctx.state === 'suspended') {
                    this.ctx.resume();
                }
            }

            playPop() {
                if (this.muted) return;
                this.init();
                if (!this.ctx) return;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, this.ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(800, this.ctx.currentTime + 0.1);
                gain.gain.setValueAtTime(0.3, this.ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.1);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.1);
            }

            playCorrect() {
                if (this.muted) return;
                this.init();
                if (!this.ctx) return;
                const now = this.ctx.currentTime;
                const notes = [523.25, 659.25, 783.99, 1046.50];
                notes.forEach((freq, idx) => {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(freq, now + idx * 0.06);
                    gain.gain.setValueAtTime(0.2, now + idx * 0.06);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + idx * 0.06 + 0.15);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start(now + idx * 0.06);
                    osc.stop(now + idx * 0.06 + 0.15);
                });
            }

            playWrong() {
                if (this.muted) return;
                this.init();
                if (!this.ctx) return;
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(180, now);
                osc.frequency.linearRampToValueAtTime(100, now + 0.25);
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start(now);
                osc.stop(now + 0.25);
            }

            playLevelUp() {
                if (this.muted) return;
                this.init();
                if (!this.ctx) return;
                const now = this.ctx.currentTime;
                const notes = [440, 554.37, 659.25, 880];
                notes.forEach((freq, idx) => {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(freq, now + idx * 0.08);
                    gain.gain.setValueAtTime(0.25, now + idx * 0.08);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + idx * 0.08 + 0.2);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start(now + idx * 0.08);
                    osc.stop(now + idx * 0.08 + 0.2);
                });
            }
        }

        function spawnCloudsForWord(wordData) {
            const container = document.getElementById('fallingArea');
            container.innerHTML = '';
            activeClouds = [];

            const options = [
                { text: wordData.correct, isCorrect: true },
                { text: wordData.correct, isCorrect: true },
                { text: wordData.correct, isCorrect: true }
            ];

            const distractors = wordData.distractors && wordData.distractors.length >= 3 
                ? wordData.distractors.slice(0, 3)
                : (wordData.distractors && wordData.distractors.length > 0 
                    ? [wordData.distractors[0], wordData.distractors[1] || wordData.distractors[0], wordData.distractors[2] || wordData.distractors[0]]
                    : ['wrong1', 'wrong2', 'wrong3']);
            
            distractors.forEach(distText => {
                options.push({ text: distText, isCorrect: false });
                options.push({ text: distText, isCorrect: false });
            });

            shuffleArray(options);

            const containerWidth = container.clientWidth || window.innerWidth || 360;
            const totalItems = options.length;
            
            const isMobile = containerWidth < 640;
            const cloudWidth = isMobile ? Math.max(54, Math.floor(containerWidth / 4.8)) : Math.min(84, Math.floor(containerWidth / 10));

            options.forEach((opt, idx) => {
                const cloudEl = document.createElement('div');
                
                let startX;
                if (isMobile) {
                    const columns = 3;
                    const colIndex = idx % columns;
                    const colWidth = containerWidth / columns;
                    startX = Math.floor(colIndex * colWidth + (colWidth - cloudWidth) / 2);
                    startX = Math.max(4, Math.min(containerWidth - cloudWidth - 4, startX + (Math.random() * 12 - 6)));
                } else {
                    const laneWidth = containerWidth / totalItems;
                    startX = Math.floor(idx * laneWidth + Math.max(0, (laneWidth - cloudWidth) / 2));
                }

                const startY = -Math.floor(Math.random() * 320 + 40 + (idx * 25));

                cloudEl.className = 'absolute cursor-pointer select-none transition-transform active:scale-95 z-10 touch-none';
                cloudEl.style.left = `${startX}px`;
                cloudEl.style.top = `${startY}px`;
                cloudEl.style.width = `${cloudWidth}px`;

                cloudEl.innerHTML = `
                    <div class="relative bg-white/95 border-2 border-indigo-200 rounded-2xl p-1 sm:p-1.5 shadow-xl text-center hover:border-indigo-400 group">
                        <div class="text-xs sm:text-base leading-none mb-0.5">☁️</div>
                        <div class="font-en font-black text-slate-800 text-[11px] sm:text-xs md:text-sm tracking-wide leading-tight break-words">${opt.text}</div>
                    </div>
                `;

                const baseSpeed = (isMobile ? 0.75 : 0.85) + Math.random() * 0.4;
                const speed = baseSpeed * fallSpeedMultiplier;

                const cloudObj = {
                    element: cloudEl,
                    x: startX,
                    y: startY,
                    speed: speed,
                    isCorrect: opt.isCorrect,
                    text: opt.text,
                    clicked: false
                };

                const triggerHandler = (e) => {
                    e.preventDefault();
                    handleCloudClick(cloudObj);
                };

                cloudEl.addEventListener('pointerdown', triggerHandler);

                container.appendChild(cloudEl);
                activeClouds.push(cloudObj);
            });
        }

        function handleCloudClick(cloudObj) {
            if (!isGameRunning || cloudObj.clicked) return;
            cloudObj.clicked = true;

            if (cloudObj.isCorrect) {
                sound.playCorrect();
                correctHits++;
                combo++;
                currentWordHits++;
                
                const gainedScore = 100 + (combo * 10) + (currentLevel * 20);
                score += gainedScore;

                createFloatingText(`+${gainedScore} (${currentWordHits}/${REQUIRED_HITS_PER_WORD})`, cloudObj.x + 20, cloudObj.y, 'text-amber-500');
                createParticleBurst(cloudObj.x + 40, cloudObj.y + 30, '#3b82f6');

                cloudObj.element.classList.add('scale-125', 'opacity-0');
                setTimeout(() => cloudObj.element.remove(), 200);

                updateWordProgressUI();

                if (correctHits % 5 === 0) {
                    currentLevel++;
                    fallSpeedMultiplier += 0.25;
                    sound.playLevelUp();
                    showSpeedBanner();
                }

                updateHUD();

                if (currentWordHits >= REQUIRED_HITS_PER_WORD) {
                    createFloatingText('✨ CLEAR! ✨', cloudObj.x + 10, cloudObj.y - 20, 'text-emerald-600 font-black text-xl');
                    
                    activeClouds.forEach(c => {
                        if (!c.clicked) {
                            c.clicked = true;
                            c.element.classList.add('opacity-0', 'transition-opacity');
                            setTimeout(() => c.element.remove(), 300);
                        }
                    });

                    setTimeout(() => {
                        if (isGameRunning) nextTargetWord();
                    }, 500);
                } else {
                    const remainingCorrect = activeClouds.filter(c => !c.clicked && c.isCorrect);
                    if (remainingCorrect.length === 0) {
                        setTimeout(() => {
                            if (isGameRunning) {
                                const activeWordIdx = (currentTargetIndex - 1 + wordList.length) % wordList.length;
                                const currentWord = wordList[activeWordIdx];
                                spawnCloudsForWord(currentWord);
                            }
                        }, 400);
                    }
                }

            } else {
                sound.playWrong();
                wrongHits++;
                combo = 0;
                lives--;

                createFloatingText('❌ 오답!', cloudObj.x + 20, cloudObj.y, 'text-rose-500');
                cloudObj.element.classList.add('animate-shake');
                cloudObj.element.style.filter = 'sepia(1) hue-rotate(-50deg) saturate(5)';

                setTimeout(() => {
                    cloudObj.element.remove();
                }, 400);

                updateHUD();

                if (lives <= 0) {
                    endGame();
                }
            }
        }

        function gameLoop() {
            if (!isGameRunning) return;

            const containerHeight = document.getElementById('gameCanvasContainer').clientHeight - 50;

            for (let i = activeClouds.length - 1; i >= 0; i--) {
                const cloud = activeClouds[i];
                if (cloud.clicked) continue;

                cloud.y += cloud.speed;
                cloud.element.style.top = `${cloud.y}px`;

                if (cloud.y >= containerHeight) {
                    cloud.clicked = true;
                    
                    if (cloud.isCorrect) {
                        sound.playWrong();
                        lives--;
                        combo = 0;
                        wrongHits++;

                        createFloatingText('⚠️ 놓쳤어요!', cloud.x + 10, containerHeight - 30, 'text-rose-600');
                        cloud.element.remove();
                        updateHUD();

                        if (lives <= 0) {
                            endGame();
                            return;
                        }

                        const remainingCorrect = activeClouds.filter(c => !c.clicked && c.isCorrect);
                        if (remainingCorrect.length === 0 && currentWordHits < REQUIRED_HITS_PER_WORD) {
                            setTimeout(() => {
                                if (isGameRunning) {
                                    const activeWordIdx = (currentTargetIndex - 1 + wordList.length) % wordList.length;
                                    const currentWord = wordList[activeWordIdx];
                                    spawnCloudsForWord(currentWord);
                                }
                            }, 400);
                        }

                    } else {
                        cloud.element.remove();
                    }
                }
            }

            activeClouds = activeClouds.filter(c => !c.clicked || c.y < containerHeight);

            animationFrameId = requestAnimationFrame(gameLoop);
        }

        function createParticleBurst(x, y, color) {
            const container = document.getElementById('fallingArea');
            for (let i = 0; i < 8; i++) {
                const p = document.createElement('div');
                p.className = 'absolute w-2.5 h-2.5 rounded-full pointer-events-none z-20';
                p.style.backgroundColor = color;
                p.style.left = `${x}px`;
                p.style.top = `${y}px`;
                container.appendChild(p);

                const angle = Math.random() * Math.PI * 2;
                const dist = 30 + Math.random() * 40;
                const tx = Math.cos(angle) * dist;
                const ty = Math.sin(angle) * dist;

                p.animate([
                    { transform: 'translate(0, 0) scale(1)', opacity: 1 },
                    { transform: `translate(${tx}px, ${ty}px) scale(0)`, opacity: 0 }
                ], {
                    duration: 500,
                    easing: 'ease-out'
                });

                setTimeout(() => p.remove(), 500);
            }
        }

        async function saveGameResult(record) {
            const statusSpinner = document.getElementById('saveStatusSpinner');
            const statusText = document.getElementById('saveStatusText');

            statusSpinner.classList.remove('hidden');
            statusText.innerText = '성적을 저장하는 중입니다...';

            localRankings.push({
                date: new Date().toLocaleDateString('ko-KR'),
                studentName: record.studentName,
                score: record.score,
                accuracy: record.accuracy,
                difficulty: record.difficulty
            });
            localRankings.sort((a, b) => b.score - a.score);
            localStorage.setItem('past_tense_rankings', JSON.stringify(localRankings));

            if (appScriptUrl) {
                try {
                    const res = await fetch(appScriptUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'text/plain' },
                        body: JSON.stringify(record)
                    });
                    const json = await res.json();
                    if (json.success) {
                        statusText.innerText = '✅ 구글 스프레드시트에 성공적으로 저장되었습니다!';
                    } else {
                        statusText.innerText = '⚠️ 로컬에 저장되었습니다 (시트 응답 오류)';
                    }
                } catch (e) {
                    statusText.innerText = '⚠️ 로컬에 저장되었습니다 (네트워크 상태 확인)';
                }
            } else {
                statusText.innerText = '💾 로컬에 성적이 저장되었습니다 (시트 URL 미설정)';
            }

            statusSpinner.classList.add('hidden');
        }

        function setRankingDifficultyFilter(filterVal, btn) {
            currentRankingFilter = filterVal;
            document.querySelectorAll('.rank-tab-btn').forEach(b => {
                b.classList.remove('active', 'bg-amber-500', 'text-white', 'shadow-sm');
                b.classList.add('bg-white', 'text-slate-700', 'border', 'border-amber-200');
            });
            btn.classList.add('active', 'bg-amber-500', 'text-white', 'shadow-sm');
            btn.classList.remove('bg-white', 'text-slate-700', 'border', 'border-amber-200');
            loadRankings();
        }

        async function loadRankings() {
            const tbody = document.getElementById('rankingTableBody');
            tbody.innerHTML = '<tr><td colspan="5" class="p-4 text-center text-slate-400">순위 데이터를 불러오는 중...</td></tr>';

            let rankings = localRankings;

            if (appScriptUrl) {
                try {
                    const res = await fetch(`${appScriptUrl}?action=getRankings`);
                    const json = await res.json();
                    if (json.success && json.rankings) {
                        rankings = json.rankings.map(r => ({
                            studentName: r[1],
                            score: r[2],
                            accuracy: r[3],
                            difficulty: r[4] || '10개'
                        }));
                    }
                } catch (e) {
                    console.log('Fetching sheet rankings failed, using local');
                }
            }

            if (currentRankingFilter !== 'all') {
                rankings = rankings.filter(r => r.difficulty === currentRankingFilter);
            }

            if (rankings.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" class="p-4 text-center text-slate-400">해당 난이도의 순위 데이터가 없습니다.</td></tr>';
                return;
            }

            tbody.innerHTML = rankings.slice(0, 30).map((item, idx) => {
                let badge = `${idx + 1}위`;
                if (idx === 0) badge = '🥇 1위';
                else if (idx === 1) badge = '🥈 2위';
                else if (idx === 2) badge = '🥉 3위';

                const diffTag = item.difficulty ? item.difficulty : '10개';

                return `
                    <tr class="hover:bg-amber-50/50">
                        <td class="p-2.5 text-center font-extrabold">${badge}</td>
                        <td class="p-2.5 font-bold text-slate-800">${item.studentName}</td>
                        <td class="p-2 text-center">
                            <span class="inline-block bg-amber-100 text-amber-800 font-extrabold text-[11px] px-2 py-0.5 rounded-md border border-amber-200">
                                ❤️ ${diffTag}
                            </span>
                        </td>
                        <td class="p-2.5 text-right font-black text-amber-600">${Number(item.score).toLocaleString()} 점</td>
                        <td class="p-2.5 text-center font-bold text-indigo-600">${item.accuracy}</td>
                    </tr>
                `;
            }).join('');
        }

        async function syncWordsFromSheet() {
            if (!appScriptUrl) {
                showToast('웹앱 URL을 먼저 저장해주세요!');
                return;
            }

            try {
                showToast('시트 단어 목록을 불러오는 중...');
                const res = await fetch(`${appScriptUrl}?action=getWords`);
                const json = await res.json();
                if (json.success && json.words.length > 0) {
                    wordList = json.words.map(r => ({
                        present: r[0],
                        correct: r[1],
                        distractors: [r[2], r[3], r[4]].filter(Boolean)
                    }));
                    renderWordTable();
                    showToast('✅ 단어 목록 업데아되었습니다!');
                }
            } catch (e) {
                showToast('단어 목록을 불러오지 못했습니다.');
            }
        }

        function copyAppsScriptCode() {
            const codeText = document.getElementById('gasCodeDisplay').innerText;
            const textarea = document.createElement('textarea');
            textarea.value = codeText;
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand('copy');
            document.body.removeChild(textarea);
            showToast('앱스크립트 코드가 복사되었습니다!');
        }

        function showToast(msg) {
            const toast = document.createElement('div');
            toast.className = 'fixed bottom-5 left-1/2 -translate-x-1/2 bg-slate-900 text-white font-bold text-xs sm:text-sm px-4 py-2.5 rounded-2xl shadow-2xl z-50 transition-all transform duration-300';
            toast.innerText = msg;
            document.body.appendChild(toast);

            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transform = 'translate(-50%, 20px)';
                setTimeout(() => toast.remove(), 300);
            }, 2500);
        }

        function shuffleArray(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
            return array;
        }
